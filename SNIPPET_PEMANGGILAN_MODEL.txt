2b) Factory model (Ollama vs Nemotron/vLLM)
File: chatbot-agent-cs/lib/llmConfig.ts

  export function createVoltAgentLanguageModel(): LanguageModel | string {
    const cfg = getResolvedLlmConfig();
    if (cfg.chatProvider === "ollama") {
      return `ollama/${cfg.chatModel}`;
    }

    const provider = createOpenAICompatible({
      name: "vllm",
      baseURL: cfg.chatApiBaseUrl,
      apiKey: cfg.chatApiKey,
      transformRequestBody: (body) => mergeOpenAiRequestBody(body, extra),
    });

    return provider.chatModel(cfg.chatModel);
  }
